;; Основные типы аргументов (lexical token types)
;; ==============================================

;; Пустой/отсутствующий аргумент
(null)      ;; Нет аргумента (используется для optional параметров)

;; Числовые типы
(number)    ;; Непосредственное числовое значение (например, 42, 0xFF)
(hex)       ;; Шестнадцатеричное число (синтаксический сахар для number)
(bin)       ;; Двоичное число (синтаксический сахар для number)
(dec)       ;; Десятичное число (синтаксический сахар для number)

;; Символьные/именованные типы
(asmsym)    ;; Символ ассемблера (имя метки, константы)
(variable)  ;; Переменная (используется в soperand для унификации)
(symbol)    ;; Общий символ (используется в parse функциях)

;; Косвенные (indirect) адресации
(imm-indirect)  ;; Косвенная адресация через непосредственное число 
                ;; (например, (42) - значение по адресу 42)
(var-indirect)  ;; Косвенная адресация через переменную/метку
                ;; (например, (label) - значение по адресу метки)
(asmsym-indirect) ;; Специальный тип для soperand макроса

;; Регистровые типы (Z80 и подобные процессоры)
;; ===========================================

;; 8-битные регистры
(reg-a)     ;; Аккумулятор A
(reg-b)     ;; Регистр B
(reg-c)     ;; Регистр C
(reg-d)     ;; Регистр D
(reg-e)     ;; Регистр E
(reg-h)     ;; Регистр H
(reg-l)     ;; Регистр L
(reg-f)     ;; Флаговый регистр F

;; 16-битные регистры
(reg-af)    ;; Регистровая пара AF
(reg-bc)    ;; Регистровая пара BC
(reg-de)    ;; Регистровая пара DE
(reg-hl)    ;; Регистровая пара HL
(reg-sp)    ;; Указатель стека SP
(reg-ix)    ;; Индексный регистр IX
(reg-iy)    ;; Индексный регистр IY
(reg-pc)    ;; Счетчик команд PC

;; Косвенная адресация через регистры
(reg-af-indirect)   ;; Косвенная через AF (редко используется)
(reg-bc-indirect)   ;; Косвенная через BC (например, LD A,(BC))
(reg-de-indirect)   ;; Косвенная через DE (например, LD A,(DE))
(reg-hl-indirect)   ;; Косвенная через HL (часто используется)
(reg-sp-indirect)   ;; Косвенная через SP
(reg-ix-indirect)   ;; Косвенная через IX со смещением
(reg-iy-indirect)   ;; Косвенная через IY со смещением

;; Специальные регистры
(reg-i)     ;; Регистр прерываний I
(reg-r)     ;; Регистр регенерации R

;; Флаги/условия
(flag-nz)   ;; Флаг "не ноль" (not zero)
(flag-z)    ;; Флаг "ноль" (zero)
(flag-nc)   ;; Флаг "нет переноса" (no carry)
(flag-c)    ;; Флаг "перенос" (carry)
(flag-po)   ;; Флаг "четность нечетная" (parity odd)
(flag-pe)   ;; Флаг "четность четная" (parity even)
(flag-p)    ;; Флаг "знак плюс" (positive)
(flag-m)    ;; Флаг "знак минус" (minus)

;; Псевдоинструкции и директивы
;; ===========================
(org)       ;; Директива установки начального адреса
(db)        ;; Define Byte (резервирование байта/байтов)
(dw)        ;; Define Word (резервирование слова/слов)
(ds)        ;; Define Space (резервирование пространства)
(equ)       ;; Определение константы (equate)

;; Специальные типы для макросов
;; ============================

;; В макросе soperand происходят преобразования:
;; (variable) → (asmsym) в первом варианте operand
;; (variable) → (number) во втором варианте operand
;;
;; Это позволяет одной инструкции работать как с метками, так и с числами
;; Например: CALL label или CALL 0x1234

;; В макросе operand могут использоваться любые из вышеперечисленных типов
;; в зависимости от архитектуры процессора

;; Пример использования в определении инструкций:
;; (operand 'ld 2 "Load instruction"
;;          :arg1 reg-a       ;; Первый аргумент - регистр A
;;          :arg2 number      ;; Второй аргумент - число
;;          :arg3 null)       ;; Третий аргумент отсутствует

;; (operand 'jp 3 "Jump instruction"  
;;          :arg1 flag-nz     ;; Условие (не ноль)
;;          :arg2 asmsym      ;; Адрес перехода (метка или число)
;;          :arg3 null);; Основные типы аргументов (lexical token types)
